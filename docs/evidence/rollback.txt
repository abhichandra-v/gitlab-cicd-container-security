# Rolling back a bad deploy with kubectl rollout undo
# run locally against the kind cluster (dev overlay) on 2026-10-13
# kubectl v1.37.1, server v1.37.0

# 1. A healthy deployment (revision 1).

$ kubectl -n orders-dev rollout status deployment/orders-api --timeout 120s

# 2. A bad release: an image tag that does not exist.

$ kubectl -n orders-dev set image deployment/orders-api orders-api=orders-api:does-not-exist
